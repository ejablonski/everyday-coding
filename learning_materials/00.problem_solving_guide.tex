\section{How to Use This Guide and Classify the Problem}
\label{sec:tree}
% ============================================================================

Within your first 60 seconds reading any problem statement, answer three questions:
\begin{enumerate}
    \item \textbf{What kind of answer is required?} Optimization, Enumeration, Decision, Counting, or Querying? (Figure~\ref{fig:triage_tree})
    \item \textbf{How large is the input $N$?} Constraints immediately reveal which complexity classes can pass. (Section~\ref{sec:size})
    \item \textbf{Which keywords point to a known pattern?} Match the symptom in Section~\ref{sec:index} and open that module.
\end{enumerate}

\begin{calloutbox}{Counting vs. Enumeration --- The Golden Rule}
If a problem asks you to \emph{produce} every valid configuration, use \textbf{Backtracking} (Module~05). If it only asks \emph{how many} exist (often modulo $10^9 + 7$), use \textbf{Combinatorics} (Module~01) or \textbf{Counting DP} (Module~06)---never generate configurations you only intend to count!
\end{calloutbox}

\section{Step 2: Check the Input Size}
\label{sec:size}
% ============================================================================

A modern CPU executes roughly $10^8$ simple operations per second. Read the constraint on $N$ \emph{backward} to see what complexity class the problem author expects:

\begingroup\small\rowcolors{2}{codebg}{white}
\begin{tabularx}{\textwidth}{>{\raggedright\arraybackslash\bfseries}p{2.7cm} >{\raggedright\arraybackslash}p{3.4cm} >{\raggedright\arraybackslash}X}
\arrayrulecolor{primaryblue}\toprule[1.2pt]
\rowcolor{primaryblue!10}
\textcolor{primaryblue}{\textbf{Input Size $N$}} & \textcolor{primaryblue}{\textbf{Feasible Complexity}} & \textcolor{primaryblue}{\textbf{What It Usually Signals}} \\
\arrayrulecolor{codeframe}\midrule[0.6pt]
$N \le 10$ & $O(N!)$ & Try every permutation (plain backtracking). \\
$N \le 15$ & $O(3^N)$ & Submask enumeration DP over all subsets. \\
$N \le 20$ & $O(2^N)$ or $O(N^2 \cdot 2^N)$ & Subset backtracking or Bitmask DP (TSP / assignment). \\
$N \le 500$ & $O(N^3)$ & Floyd-Warshall, Interval DP, or three nested loops. \\
$N \le 5{,}000$ & $O(N^2)$ & 2D DP tables (LCS, Edit Distance) or pairwise checks. \\
$N \le 10^6$ & $O(N \log N)$ or $O(N)$ & Sorting, Greedy, Binary Search, Sieves, BFS/DFS, Mono Stack. \\
$N \le 10^8$ & $O(N)$ (tiny constant) & Single linear pass, bit operations, or simple sieve. \\
$N \ge 10^9$ & $O(\log N)$ or $O(1)$ & Closed-form formula, Binary Search, Fast Exp, or $O(\sqrt{N})$. \\
\arrayrulecolor{primaryblue}\bottomrule[1.2pt]
\end{tabularx}\endgroup

\section{Glossary of Terms and Symbols}
% ============================================================================

\subsection*{Core Concepts}
\begin{itemize}[itemsep=2pt]
    \item \textbf{Optimization / Enumeration / Decision / Counting / Querying}: The five core output categories in Figure~\ref{fig:triage_tree}.
    \item \textbf{Precomputation}: Paying $O(N)$ build time once so every later query runs in $O(1)$ or $O(\log N)$.
    \item \textbf{Backtracking vs.\ Dynamic Programming vs.\ Greedy}: Backtracking explores and undoes all valid branches; DP caches overlapping subproblems; Greedy commits irreversibly via an Exchange Argument.
    \item \textbf{Monotonicity}: A predicate that flips truth value at most once ($\texttt{FFFTTT}$), enabling Binary Search on the Answer.
    \item \textbf{Disjoint Set Union (DSU)}: Near-$O(1)$ ($O(\alpha(N))$) component union/find using path compression + union by rank.
    \item \textbf{Monotonic Stack / Deque}: Sorted stack or double-ended queue maintaining nearest extrema in $O(N)$ amortized time.
    \item \textbf{Trie / Double Rolling Hash / Border}: Core string primitives for $O(L)$ prefix lookup, $O(1)$ slice equality, and KMP search.
    \item \textbf{Bitmask / Two's Complement / Self-Inverse}: Subset integer encoding, negative-MSB signed arithmetic, and $x \oplus x = 0$.
    \item \textbf{Two Pointers / Sliding Window / Floyd's Cycle}: Monotonic linear scans ($O(N)$), $\text{atMost}(K) - \text{atMost}(K{-}1)$ counting, and $O(1)$-space cycle entry.
    \item \textbf{Binary Heap / Floyd Heapify / Lazy Deletion}: Complete-tree flat array ($2i{+}1, 2i{+}2$), $O(N)$ bottom-up build, Two-Heap median, and tombstone pruning.
    \item \textbf{Euler Tour / Tree Diameter / Binary Lifting / BST}: Subtree-to-range flattening ($[\text{tin}, \text{tout}]$), longest tree path, $2^k$-ancestor LCA, and global ordered search trees.
    \item \textbf{Fenwick Tree (BIT) / Segment Tree / Lazy Propagation}: $1$-based lowbit prefix buckets ($i \mathbin{\&} -i$), $2N$/$4N$ monoid range trees, and deferred range-update tags (\texttt{apply}/\texttt{push}/\texttt{pull}).
\end{itemize}

\subsection*{Mathematical Symbols}
\begin{itemize}[itemsep=2pt]
    \item $N, V, E, W$: Input size, graph vertices/edges, and knapsack weight capacity.
    \item $O(\cdot),\; \log N,\; \sqrt{N},\; \alpha(N)$: Complexity growth, binary logarithm, square root, and inverse Ackermann ($\le 4$).
    \item $\sum,\; \prod,\; \lfloor \cdot \rfloor,\; \lceil \cdot \rceil$: Summation, product, floor, and ceiling functions.
    \item $N!,\; \binom{n}{k},\; D_n,\; C_n,\; S(n,k)$: Factorial, binomial coefficient, derangements, Catalan numbers, and Stirling numbers of the second kind.
    \item $\bmod,\; \gcd,\; \operatorname{lcm},\; \phi(N)$: Remainder, greatest common divisor, least common multiple, and Euler's totient.
    \item $\oplus,\; \mathbin{\&},\; \pi[i],\; Z[i],\; \Sigma,\; \mu,\; \lambda$: Bitwise XOR/AND, KMP $\pi$-array, Z-array, alphabet size, and Floyd's tail/cycle lengths.
    \item $\text{tin}[v],\; \text{tout}[v],\; \operatorname{lca}(u, v)$: Euler Tour DFS timestamps and Lowest Common Ancestor in a rooted tree.
    \item $\operatorname{lowbit}(i),\; \text{bit}[i],\; (S, \oplus, e),\; \text{lazy}[v]$: Lowest set bit $i \mathbin{\&} (-i)$, Fenwick bucket, associative monoid with identity $e$, and deferred Segment Tree tag.
\end{itemize}

\section*{Solutions to Pen-and-Paper Exercises}
\addcontentsline{toc}{section}{Solutions to Pen-and-Paper Exercises}
\markboth{Solutions to Pen-and-Paper Exercises}{}

\begin{enumerate}[itemsep=3pt]
    \item \label{sol:ex1}\textbf{Exercise 1 (Triage) --- Constraints-Driven Triage:}
    \emph{P1}: Optimization; $O(N \cdot 2^N)$; \textbf{Bitmask DP} (Module~06 / Module~12).
    \emph{P2}: Querying; $O(N)$; \textbf{Monotonic Deque} (Module~11).
    \emph{P3}: Counting; $O((R{-}L)\log\log R)$; \textbf{Segmented Sieve} (Module~03).
    \trapnote{Trying a standard Sieve of Eratosthenes on $R = 10^{11}$ requires $100\text{ GB}$ of RAM---always check window width $R - L$ when $R \ge 10^9$.}

    \item \label{sol:ex2}\textbf{Exercise 2 (Triage) --- Greedy, DP, or Binary Search:}
    \emph{P4}: Optimization / Decision; $O(N \log(\sum w_i))$; \textbf{Binary Search on the Answer} + greedy day-packing check (Module~08).
    \emph{P5}: Optimization; $O(N \log N)$; \textbf{Interval Scheduling Greedy} sorted by finish time $f_i$ (Module~10).
    \emph{P6}: Optimization; $O(N \log N)$; \textbf{Weighted Interval DP} (sort by $f_i$ + binary search predecessor; Module~06 + Module~08).
    \trapnote{Assuming earliest-finish greedy still works once talks carry unequal profits---arbitrary weights break the Exchange Argument and force DP.}

    \item \label{sol:ex3}\textbf{Exercise 3 (Triage) --- Strings, Graphs, and Counting Disguises:}
    \emph{P7}: Decision / Ordering; $O(V + E)$; \textbf{Kahn's Topological Sort} (Module~07).
    \emph{P8}: Querying / Structure; $O(N)$; \textbf{KMP Prefix Function} checking $p = N - \pi[N-1]$ (Module~09).
    \emph{P9}: Counting; $O(N)$ / $O(1)$; \textbf{Stars and Bars} with lower bounds $\binom{50-1}{8-1} = \binom{49}{7} \bmod (10^9{+}7)$ (Module~01 + Module~02).
    \trapnote{Forgetting the lower-bound shift in Stars and Bars (pre-placing $1$ coin in each of the $8$ jars reduces $n$ from $50$ to $42$, giving $\binom{42 + 8 - 1}{8 - 1} = \binom{49}{7}$).}
\end{enumerate}
