% ============================================================================
\section{Foundations}
% ============================================================================

Ordinary arithmetic stretches forever along an infinite number line. Modular arithmetic bends that line into a finite circle so numbers wrap smoothly back to $0$ every $m$ steps.

\subsection{Clock Arithmetic and Congruence}

\begin{notationbox}[Symbols Used in Foundations]
$m$ (or $n$) & Positive integer \textbf{modulus} where numbers wrap around & $m = 12$ or $m = 10^9 + 7$ \\
$a \equiv b \pmod m$ & ``$a$ is congruent to $b$ modulo $m$'' (same remainder) & $14 \equiv 2 \pmod{12}$ \\
$m \mid (a - b)$ & ``$m$ divides $(a - b)$ evenly'' (zero remainder) & $12 \mid (14 - 2)$ \\
$\mathbb{Z}/m\mathbb{Z}$ & Closed ring of the $m$ remainders $\{0, 1, \dots, m - 1\}$ & $\mathbb{Z}/12\mathbb{Z} = \{0, \dots, 11\}$ \\
\end{notationbox}

Think of a standard 12-hour clock where $12$ is labeled $0$. If it is $9{:}00$ and $5$ hours pass, you land on $2{:}00$ rather than $14{:}00$ because $14$ wraps past $12$ with remainder $2$. Moving backwards $3$ hours from $0{:}00$ lands on $9{:}00$, so $-3 \equiv 9 \pmod{12}$.

\begin{formulabox}[Congruence Modulo $m$ in Plain English]
\[
    a \equiv b \pmod m \quad \Longleftrightarrow \quad m \mid (a - b) \quad \Longleftrightarrow \quad \text{rem}(a, m) = \text{rem}(b, m)
\]
\textbf{Read this as:} Two integers $a$ and $b$ are congruent modulo $m$ when they sit at the exact same spot on an $m$-hour clock---equivalently, their difference $a - b$ is an exact multiple of $m$.
\end{formulabox}

Every modulus $m$ partitions all integers into exactly $m$ disjoint buckets called \textbf{residue classes}: $[r] = \{r + k \cdot m \mid k \in \mathbb{Z}\}$. Two numbers in the same bucket behave identically under $+, -, \times$ modulo $m$.

\subsection{Why Modular Arithmetic Matters in Programming}

Modular arithmetic shows up in software engineering and algorithmic problem solving in three distinct ways:
\begin{enumerate}
    \item \textbf{Answers too huge to store:} Counting and DP problems routinely produce values with thousands of digits. Reporting the answer modulo $M = 10^9 + 7$ keeps every intermediate number small by reducing after every step.
    \item \textbf{Cyclic / wrap-around indexing:} Ring buffers, round-robin load balancers, and circular arrays advance via \texttt{next = (cur + step) \% N}---direct modulo with no heavy machinery.
    \item \textbf{Modular division:} Whenever a formula divides (such as $\binom{n}{k} = \frac{n!}{k!(n-k)!}$), raw integer division \texttt{(a / b) \% M} is mathematically broken; you must multiply by $b$'s \textbf{modular inverse} (Section~\ref{sec:inverses}).
\end{enumerate}

\begin{calloutbox}{Why Specifically $M = 10^9 + 7$ (and $998{,}244{,}353$)?}
Three engineering properties make $M = 1{,}000{,}000{,}007$ the universal default:
\begin{itemize}
    \item \textbf{Prime:} Every nonzero value $1 \dots M - 1$ has a unique modular multiplicative inverse.
    \item \textbf{32-bit safe addition:} $2M = 2{,}000{,}000{,}014 < 2^{31} - 1 = 2{,}147{,}483{,}647$, so adding two reduced values fits inside a signed 32-bit \texttt{int}.
    \item \textbf{64-bit safe multiplication:} $M^2 \approx 10^{18} < 2^{63} - 1 \approx 9.22 \times 10^{18}$, so multiplying two reduced values fits inside a signed 64-bit \texttt{std::int64\_t} before applying \texttt{\% M}.
\end{itemize}
\end{calloutbox}

\]
\end{tryitbox}

\subsection{Why $+, -, \times$ Preserve Remainders --- But Division Breaks}

\painbeat Suppose you must compute $(10^8 \times 10^8 + 10^8) \bmod 7$. Computing the full product first overflows a 32-bit integer (and multiplying ten such numbers overflows even 64-bit \texttt{int64\_t}).

\ahabeat Write any two integers as multiples of $m$ plus their remainders: $a = q_1 m + r_1$ and $b = q_2 m + r_2$. When you add, subtract, or multiply $a$ and $b$, every term containing a factor of $m$ disappears modulo $m$, leaving only $r_1 + r_2$, $r_1 - r_2$, or $r_1 \cdot r_2$!

\examplebeat Test on $a = 17$ ($r_1 = 3$) and $b = 10$ ($r_2 = 3$) modulo $m = 7$:
\begin{itemize}
    \item Product first: $(17 \times 10) \bmod 7 = 170 \bmod 7 = 2$.
    \item Reduce first: $((17 \bmod 7) \times (10 \bmod 7)) \bmod 7 = (3 \times 3) \bmod 7 = 9 \bmod 7 = 2$. Exact match!
\end{itemize}

\begin{formulabox}[Reduce-As-You-Go Rules (Homomorphic Compatibility)]
If $a \equiv r_1 \pmod m$ and $b \equiv r_2 \pmod m$, then:
\[
\begin{aligned}
    (a + b) \bmod m &= (r_1 + r_2) \bmod m, \\
    (a - b) \bmod m &= (r_1 - r_2 + m) \bmod m, \\
    (a \cdot b) \bmod m &= (r_1 \cdot r_2) \bmod m.
\end{aligned}
\]
\textbf{Read this as:} You may safely take \texttt{\% m} after \emph{every} addition, subtraction, or multiplication without ever changing the final remainder.
\end{formulabox}

\begin{watchoutbox}[Watch Out --- Why Raw Division Breaks Immediately]
Try $(6 / 2) \bmod 4$: the true integer quotient is $3$, so $(6 / 2) \bmod 4 = \mathbf{3}$. Yet if you reduce the numerator first, $6 \equiv 2 \pmod 4$, and $(2 / 2) \bmod 4 = \mathbf{1} \neq 3$! Worse, for $(10 / 2) \bmod 7 = 5$, reducing first gives $3 / 2$, which is not even an integer! \textbf{Never divide directly after taking a modulo}---always multiply by a modular inverse (Section~\ref{sec:inverses}).
\end{watchoutbox}

\begin{tryitbox}{Debug}{Spotting Modular Overflow and Broken Division}{ex2}
A programmer wants $(a \cdot b) / c \pmod M$ with $M = 10^9 + 7$, $a = 10^9$, $b = 10^9$, $c = 2$ (knowing $2 \mid ab$), and writes:
\begin{lstlisting}[numbers=none, xleftmargin=4pt, framexleftmargin=4pt]
int M = 1'000'000'007;
int prod = (a * b) % M;   // Bug 1
int ans  = (prod / c) % M; // Bug 2
\end{lstlisting}
Explain why \textbf{both} lines fail and state the exact true value of $(a \cdot b / 2) \bmod (10^9 + 7)$ when $a = b = 10^9$ (hint: $10^9 \equiv -7 \pmod M$).
\end{tryitbox}

\subsection{Modular Binary Exponentiation ($O(\log b)$)}

\painbeat Computing $a^b \bmod m$ by multiplying $a$ into an accumulator $b$ times takes $O(b)$ steps. When $b = 10^{18}$, a naive loop would run for centuries.

\ahabeat Write the exponent $b$ in \textbf{binary}! Every exponent is a sum of distinct powers of two, so $a^b$ splits into a product of repeated squares $a^1, a^2, a^4, a^8, \dots$---each square costs just \emph{one} multiplication from the previous square!

\examplebeat For $a^{13}$, note $13 = 8 + 4 + 1 = (1101)_2$. Therefore:
\[
    a^{13} = a^{8 + 4 + 1} = a^8 \cdot a^4 \cdot a^1.
\]
Instead of $12$ multiplications, we compute $a \to a^2 \to a^4 \to a^8$ (3 squarings) and multiply the three active powers corresponding to the $1$-bits of $13$ (3 multiplications)---6 operations total! For $b = 10^{18}$, $\lfloor \log_2(10^{18}) \rfloor = 59$, so at most $119$ multiplications finish the entire job!

\begin{formulabox}[Binary Exponentiation Recurrence]
\[
    a^b \bmod m =
    \begin{cases}
        1 \bmod m & \text{if } b = 0, \\
        \bigl(a^{b/2} \bmod m\bigr)^2 \bmod m & \text{if } b \text{ is even}, \\
        \bigl(a \cdot (a^{(b-1)/2} \bmod m)^2\bigr) \bmod m & \text{if } b \text{ is odd}.
    \end{cases}
\]
\textbf{Read this as:} Inspect the lowest bit of $b$. If it is $1$, multiply \texttt{result} by \texttt{base}; then square \texttt{base} and shift $b$ right by $1$ bit, reducing modulo $m$ on every single multiplication.
\end{formulabox}

\subsection{Modular Inverses: Extended Euclid vs.\ Fermat's Little Theorem}\label{sec:inverses}

\begin{notationbox}[Symbols Used in Modular Inverses]
$a^{-1} \bmod m$ & Integer $x \in [0, m - 1]$ satisfying $a \cdot x \equiv 1 \pmod m$ & $3^{-1} \equiv 5 \pmod 7$ ($3 \times 5 = 15 \equiv 1$) \\
$\gcd(a, m)$ & Greatest common divisor of $a$ and $m$ & $\gcd(a, m) = 1 \iff$ inverse exists \\
$x, y$ & B\'{e}zout integers satisfying $a x + m y = \gcd(a, m)$ & $17(-5) + 43(2) = 1$ \\
$p$ & Prime modulus (required for Fermat's Little Theorem) & $p = 10^9 + 7$ \\
\end{notationbox}

\painbeat To compute $(a / b) \bmod m$, you cannot divide directly. Searching every candidate $x \in \{1, \dots, m - 1\}$ until $b \cdot x \equiv 1 \pmod m$ takes $O(m)$ time---far too slow when $m = 10^9 + 7$.

\ahabeat Division by $b$ is just multiplication by $b^{-1}$. When $\gcd(b, m) = 1$, there are two $O(\log m)$ ways to compute $b^{-1} \bmod m$:
\begin{enumerate}
    \item \textbf{Fermat's Little Theorem (Prime $p$ only):} Multiply the set $\{1, 2, \dots, p - 1\}$ by $a$ modulo $p$. Because $p$ is prime and $p \nmid a$, every product is distinct and nonzero---it is the \emph{exact same set} rearranged! Multiplying all $p - 1$ numbers together gives $(p - 1)! \equiv a^{p-1}(p - 1)! \pmod p$. Canceling $(p - 1)!$ yields $a^{p-1} \equiv 1 \pmod p$, so $a \cdot a^{p-2} \equiv 1 \pmod p$!
    \item \textbf{Extended Euclidean Algorithm (Any modulus $m$ with $\gcd(a, m) = 1$):} By B\'{e}zout's Identity, Euclid's division steps can be reversed to find integers $x, y$ with $a x + m y = \gcd(a, m)$. When $\gcd(a, m) = 1$, taking both sides modulo $m$ wipes out $m y$, leaving $a x \equiv 1 \pmod m$!
\end{enumerate}

\examplebeat Test Fermat's permutation proof on $p = 5$ and $a = 2$:
\[
    \{1, 2, 3, 4\} \xrightarrow{\times 2 \bmod 5} \{2, 4, 1, 3\} \quad (\text{exact same set!}).
\]
So $2^{5 - 1} = 16 \equiv 1 \pmod 5$, and $2^{-1} \equiv 2^{5 - 2} = 8 \equiv \mathbf{3} \pmod 5$ (check: $2 \times 3 = 6 \equiv 1$).

\begin{formulabox}[Two $O(\log m)$ Modular Inverse Methods]
\[
\begin{aligned}
    \text{Prime } p \text{ (Fermat):}\quad & a^{-1} \equiv a^{p - 2} \pmod p \\
    \text{Any coprime } m \text{ (Extended Euclid):}\quad & a x + m y = 1 \implies a^{-1} \equiv x \pmod m
\end{aligned}
\]
\textbf{Read this as:} Use Fermat ($a^{p-2} \bmod p$) when $p$ is prime---it is 5 lines using \texttt{mod\_pow}. Use Extended Euclid whenever $m$ is composite or unknown, as long as $\gcd(a, m) = 1$.
\end{formulabox}

\begin{watchoutbox}[Watch Out --- Two Ways Inverses Fail]
\begin{itemize}
    \item \textbf{Non-coprime inputs ($\gcd(a, m) > 1$):} No inverse exists at all! For example, $2x \bmod 6$ only ever hits $\{0, 2, 4\}$---it can never equal $1$.
    \item \textbf{Fermat on a composite modulus:} Never compute $a^{m - 2} \bmod m$ when $m$ is composite! For $a = 3, m = 10$, $3^{8} = 6561 \equiv 1 \pmod{10}$, yet $3 \times 1 \not\equiv 1 \pmod{10}$ (the real inverse is $7$).
\end{itemize}
\end{watchoutbox}

\begin{enumerate}
    \item Find $5^{-1} \bmod 11$ using the Extended Euclidean Algorithm (write the forward division and back-substitution into $5x + 11y = 1$).
    \item Confirm your answer using Fermat's Little Theorem by computing $5^9 \bmod 11$ via repeated squaring.
\end{enumerate}
\end{tryitbox}

\subsection{Chinese Remainder Theorem (Aligning Periodic Clocks)}

\begin{notationbox}[Symbols Used in CRT and Euler's Theorem]
$n_1, \dots, n_k$ & Pairwise coprime moduli ($\gcd(n_i, n_j) = 1$ for $i \neq j$) & $n_1 = 3,\; n_2 = 5,\; n_3 = 7$ \\
$N$ & Product of all moduli ($N = \prod_{i=1}^{k} n_i$) & $N = 3 \times 5 \times 7 = 105$ \\
$M_i$ & Partial product excluding $n_i$ ($M_i = N / n_i$) & $M_1 = 35,\; M_2 = 21,\; M_3 = 15$ \\
$\phi(n)$ & Euler's Totient: count of integers in $[1, n]$ coprime to $n$ & $\phi(12) = 4$ ($\{1, 5, 7, 11\}$) \\
\end{notationbox}

\painbeat Suppose three independent periodic clocks (a 3-hour, a 5-hour, and a 7-hour clock) currently read $2$, $3$, and $2$. What time $x$ aligns all three clocks at once? Brute-forcing $x = 0, 1, 2, \dots$ fails completely if the product of moduli is $10^{18}$.

\ahabeat Build orthogonal ``magic switches''! For each clock $i$, let $M_i = N / n_i$ be the product of all \emph{other} moduli, so $M_i \equiv 0 \pmod{n_j}$ for every $j \neq i$. Multiplying $M_i$ by its inverse $M_i^{-1} \bmod n_i$ makes the switch $e_i = M_i \cdot M_i^{-1}$ equal $1 \pmod{n_i}$ and $0$ modulo every other clock! Summing $\sum a_i e_i$ lets every clock see \emph{only} its own target remainder $a_i$!

\begin{formulabox}[Chinese Remainder Theorem (Pairwise Coprime Moduli)]
For pairwise coprime $n_1, \dots, n_k$ with $N = \prod_{i=1}^{k} n_i$ and $M_i = N / n_i$, the system $x \equiv a_i \pmod{n_i}$ has a \textbf{unique} solution modulo $N$:
\[
    x \equiv \sum_{i=1}^{k} a_i \cdot M_i \cdot (M_i^{-1} \bmod n_i) \pmod N.
\]
\textbf{Read this as:} Each switch $M_i \cdot M_i^{-1}$ equals $1$ on clock $i$ and $0$ on all other clocks. Scale switch $i$ by $a_i$, add them up, and reduce modulo $N$.
\end{formulabox}

\begin{watchoutbox}[Watch Out --- Non-Coprime Moduli Can Have Zero Solutions]
Basic CRT requires pairwise coprime moduli ($\gcd(n_i, n_j) = 1$). If two moduli share a factor $g = \gcd(n_1, n_2) > 1$, a solution exists \emph{only} when $a_1 \equiv a_2 \pmod g$ (for example, $x \equiv 1 \pmod 4$ and $x \equiv 2 \pmod 6$ is impossible: $x$ cannot be both odd and even!).
\end{watchoutbox}

\subsection{Euler's Totient Function and Euler's Theorem}

\painbeat What if a problem asks for $a^B \bmod n$ where $n$ is \emph{composite} (say $n = 100$) or the exponent $B$ itself is a tower like $2^{10^{18}}$? Fermat's Little Theorem cannot help because $n$ is not prime.

\ahabeat Repeat Fermat's permutation trick, keeping \emph{only} the $\phi(n)$ numbers in $\{1, \dots, n\}$ that are coprime to $n$! When $\gcd(a, n) = 1$, multiplying those $\phi(n)$ numbers by $a$ modulo $n$ permutes them into the exact same set, proving $a^{\phi(n)} \equiv 1 \pmod n$!

\begin{formulabox}[Euler's Totient $\phi(n)$ and Euler's Theorem]
\[
\begin{aligned}
    \phi(n) &= n \prod_{p \mid n} \left(1 - \frac{1}{p}\right) \\
    \text{When } \mathbf{\gcd(a, n) = 1}:\quad a^{\phi(n)} &\equiv 1 \pmod n \implies a^B \equiv a^{B \bmod \phi(n)} \pmod n
\end{aligned}
\]
\textbf{Read this as:} $\phi(n)$ counts how many integers from $1$ to $n$ share no factor with $n$ (for prime $p$, $\phi(p) = p - 1$; for prime power $p^k$, $\phi(p^k) = p^{k-1}(p - 1)$). Whenever $a$ and $n$ are coprime, powers of $a$ reset every $\phi(n)$ steps!
\end{formulabox}

\begin{watchoutbox}[Watch Out --- Exponent Reduction Strictly Requires $\gcd(a, n) = 1$!]
Reducing $a^B \bmod n$ to $a^{B \bmod \phi(n)} \bmod n$ is \textbf{invalid} whenever $\gcd(a, n) > 1$!
\begin{itemize}
    \item \textbf{Counterexample:} Take $a = 2, n = 4, B = 4$. Here $\phi(4) = 2$ and $4 \bmod 2 = 0$. The naive shortcut gives $2^0 = 1 \pmod 4$, whereas the true value is $2^4 = 16 \equiv \mathbf{0} \pmod 4$!
    \item \textbf{Generalized rule (when $\gcd(a, n) > 1$ and $B \ge \phi(n)$):} Shift the exponent into the repeating tail via $a^B \equiv a^{\phi(n) + (B \bmod \phi(n))} \pmod n$.
\end{itemize}
\end{watchoutbox}

\end{enumerate}

\subsection{Real-World Systems Applications}

Modular arithmetic quietly powers everyday engineering infrastructure:
\begin{itemize}
    \item \textbf{Public-Key Cryptography (RSA \& Diffie-Hellman):} RSA encrypts via $C = M^e \bmod n$ and decrypts via $M = C^d \bmod n$ where $d = e^{-1} \bmod \phi(n)$. Diffie-Hellman exchanges $g^a \bmod p$ and $g^b \bmod p$ to agree on $g^{ab} \bmod p$ without ever sending $a$ or $b$.
    \item \textbf{Systems \& Data Structures:} Ring buffers advance via \texttt{(tail + 1) \% N}, hash tables map keys via \texttt{hash \% capacity}, round-robin schedulers route via \texttt{id \% K}, and linear congruential PRNGs step via $X_{n+1} = (a X_n + c) \bmod m$.
\end{itemize}

\arrayrulecolor{codeframe}\midrule[0.6pt]
Wrap-Around Indexing & Raw index grows out of bounds or goes negative & \texttt{((idx + step) \% N + N) \% N} in $O(1)$ \\

\begin{calloutbox}{Pattern Bonus: Tracking Remainders via Pigeonhole Cycles}
Some problems ask for the smallest number made only of repeated digits (like repunits $1, 11, 111, \dots$) divisible by $k$. You never build the giant decimal string---just update \texttt{rem = (rem * 10 + 1) \% k}! Because there are only $k$ possible remainders $\{0, \dots, k - 1\}$, by the Pigeonhole Principle you either hit $0$ or enter a cycle within at most $k$ steps!
\end{calloutbox}

Write $19 = 16 + 2 + 1 = (10011)_2$ and trace from the least-significant bit ($b_0$ to $b_4$) with $\text{res} = 1,\; \text{base} = 2$:
\begin{enumerate}
    \item \textbf{Bit $b_0 = 1$ ($\text{exp} = 19$):} $\text{res} = (1 \times 2) \bmod 13 = 2$;\; $\text{base} = 2^2 \bmod 13 = 4$;\; $\text{exp} = 9$.
    \item \textbf{Bit $b_1 = 1$ ($\text{exp} = 9$):} $\text{res} = (2 \times 4) \bmod 13 = 8$;\; $\text{base} = 4^2 \bmod 13 = 3$;\; $\text{exp} = 4$.
    \item \textbf{Bit $b_2 = 0$ ($\text{exp} = 4$):} $\text{res}$ stays $8$;\; $\text{base} = 3^2 \bmod 13 = 9$;\; $\text{exp} = 2$.
    \item \textbf{Bit $b_3 = 0$ ($\text{exp} = 2$):} $\text{res}$ stays $8$;\; $\text{base} = 9^2 \bmod 13 = 81 \bmod 13 = 3$;\; $\text{exp} = 1$.
    \item \textbf{Bit $b_4 = 1$ ($\text{exp} = 1$):} $\text{res} = (8 \times 3) \bmod 13 = 24 \bmod 13 = \mathbf{11}$;\; $\text{exp} = 0$ (done).
\end{enumerate}
\textbf{Check via Fermat:} $2^{12} \equiv 1 \pmod{13} \implies 2^{19} \equiv 2^7 = 128 = 9 \times 13 + 11 \equiv \mathbf{11} \pmod{13}$.
\end{example}

\begin{example}[Extended Euclidean Algorithm with Complete Back-Substitution: Find $17^{-1} \bmod 43$]
We seek $x$ such that $17x \equiv 1 \pmod{43}$ (equivalently $17x + 43y = 1$).
\begin{enumerate}
    \item \textbf{Forward Euclidean divisions} (isolate each remainder):
    \begin{align*}
        43 &= 2 \times 17 + 9 &&\implies 9 = 43 - 2 \times 17 \\
        17 &= 1 \times 9 + 8  &&\implies 8 = 17 - 1 \times 9 \\
        9  &= 1 \times 8 + 1  &&\implies 1 = 9 - 1 \times 8 \quad (\gcd = 1,\text{ inverse exists!})
    \end{align*}
    \item \textbf{Backward substitution} (start from $1$ and eliminate $8$, then $9$):
    \begin{align*}
        1 &= 9 - 1 \times 8 \\
          &= 9 - 1 \times (17 - 1 \times 9) = 2 \times 9 - 1 \times 17 \\
          &= 2 \times (43 - 2 \times 17) - 1 \times 17 = 2 \times 43 - 5 \times 17.
    \end{align*}
    \item \textbf{Reduce modulo $43$:} $-5 \times 17 \equiv 1 \pmod{43}$, so $17^{-1} \equiv -5 \equiv -5 + 43 = \mathbf{38} \pmod{43}$.
    \item \textbf{Verify:} $17 \times 38 = 646 = 15 \times 43 + 1 \equiv 1 \pmod{43}$.
\end{enumerate}
\end{example}

\begin{example}[Chinese Remainder Theorem (Three Congruences)]
Solve $x \equiv 2 \pmod 3,\; x \equiv 3 \pmod 5,\; x \equiv 2 \pmod 7$.
\begin{enumerate}
    \item Pairwise coprime check: $\gcd(3, 5) = \gcd(3, 7) = \gcd(5, 7) = 1$; product $N = 3 \times 5 \times 7 = 105$.
    \item Compute partial products $M_i = N / n_i$ and their inverses $M_i^{-1} \bmod n_i$:
    \begin{itemize}
        \item $M_1 = 35 \equiv 2 \pmod 3 \implies 35^{-1} \equiv 2 \pmod 3$ ($35 \times 2 = 70 \equiv 1$).
        \item $M_2 = 21 \equiv 1 \pmod 5 \implies 21^{-1} \equiv 1 \pmod 5$ ($21 \times 1 = 21 \equiv 1$).
        \item $M_3 = 15 \equiv 1 \pmod 7 \implies 15^{-1} \equiv 1 \pmod 7$ ($15 \times 1 = 15 \equiv 1$).
    \end{itemize}
    \item Combine the scaled switches (matching Figure~\ref{fig:crt-clocks}):
    \[
        x \equiv 2(35)(2) + 3(21)(1) + 2(15)(1) = 140 + 63 + 30 = 233 \equiv \mathbf{23} \pmod{105}.
    \]
\end{enumerate}
\end{example}

\begin{example}[Smallest Repunit Divisible by $k = 7$]
Find the length of the shortest all-ones decimal number ($1, 11, 111, \dots$) divisible by $7$.
\begin{enumerate}
    \item Since $7$ is divisible by neither $2$ nor $5$, a solution is guaranteed within $7$ steps.
    \item Track $r \leftarrow (10r + 1) \bmod 7$:
    \begin{itemize}
        \item Length 1: $1 \bmod 7 = 1$;\quad Length 2: $11 \bmod 7 = 4$;\quad Length 3: $41 \bmod 7 = 6$;
        \item Length 4: $61 \bmod 7 = 5$;\quad Length 5: $51 \bmod 7 = 2$;\quad Length 6: $21 \bmod 7 = \mathbf{0}$!
    \end{itemize}
    \item Answer: length \textbf{6} (indeed $111{,}111 = 7 \times 15{,}873$).
\end{enumerate}
\end{example}

Every listing below compiles cleanly under \texttt{g++ -std=c++23 -Wall -Wextra}, normalizes negative remainders safely, and guards invalid inputs.

\subsection{Safe Modulo and Standalone Binary Exponentiation}

Rather than writing \texttt{((a \% m) + m) \% m} (which runs two divisions and can overflow \texttt{(a \% m) + m} if \texttt{a >= 0} and \texttt{m > INT64\_MAX / 2}), we compute \texttt{r = a \% m} once and add \texttt{m} only when \texttt{r < 0}. Notice that \texttt{mod\_pow} also handles \texttt{mod == 1} (where every number is congruent to $0$) and skips the unused final squaring.

\begin{lstlisting}[caption={Safe Canonical Modulo and Standalone O(log exp) Binary Exponentiation}]
#include <cstdint>
#include <stdexcept>

// Returns the canonical remainder of a modulo m in [0, m - 1].
// Avoids double division and avoids (r + m) overflow when r >= 0.
[[nodiscard]] constexpr std::int64_t safe_mod(std::int64_t a, std::int64_t m) {
    if (m <= 0) throw std::invalid_argument("Modulus must be strictly positive");
    std::int64_t r = a % m;
    return (r < 0) ? (r + m) : r;
}

// Computes (base^exp) % mod in O(log exp) time.
// Requires 0 < mod <= 4'294'967'295 so uint64_t products never overflow.
[[nodiscard]] constexpr std::int64_t mod_pow(
        std::int64_t base, std::uint64_t exp, std::int64_t mod) {
    if (mod <= 0) throw std::invalid_argument("Modulus must be strictly positive");
    if (mod == 1) return 0;

    std::uint64_t m = static_cast<std::uint64_t>(mod);
    std::uint64_t b = static_cast<std::uint64_t>(safe_mod(base, mod));
    std::uint64_t result = 1;

    while (exp > 0) {
        if (exp & 1ULL) result = (result * b) % m;
        exp >>= 1;
        if (exp > 0) b = (b * b) % m; // Skip unused final squaring
    }
    return static_cast<std::int64_t>(result);
}
\end{lstlisting}

\subsection{Standalone Extended Euclidean Algorithm and Chinese Remainder Theorem}

\texttt{mod\_inverse} returns \texttt{std::optional<std::int64\_t>} (empty when $\gcd(a, m) > 1$). \texttt{solve\_crt} merges congruences pairwise in $O(k \log N)$ time without exceptions.

\begin{lstlisting}[caption={Standalone Extended Euclid, Modular Inverse, and Pairwise CRT Solver}]
#include <cstdint>
#include <optional>
#include <span>
#include <utility>

struct GcdResult {
    std::int64_t g; // gcd(a, b) >= 0
    std::int64_t x; // Bezout coefficient for a
    std::int64_t y; // Bezout coefficient for b: a*x + b*y = g
};

// Iterative Extended Euclidean Algorithm: O(log min(|a|, |b|)) time, O(1) space.
[[nodiscard]] constexpr GcdResult ext_gcd(std::int64_t a, std::int64_t b) noexcept {
    std::int64_t old_r = a, r = b;
    std::int64_t old_x = 1, x = 0;
    std::int64_t old_y = 0, y = 1;
    while (r != 0) {
        std::int64_t q = old_r / r;
        old_r = std::exchange(r, old_r - q * r);
        old_x = std::exchange(x, old_x - q * x);
        old_y = std::exchange(y, old_y - q * y);
    }
    if (old_r < 0) {
        old_r = -old_r;
        old_x = -old_x;
        old_y = -old_y;
    }
    return {old_r, old_x, old_y};
}

// Returns a^(-1) mod m in [0, m - 1], or std::nullopt if gcd(a, m) != 1.
[[nodiscard]] constexpr std::optional<std::int64_t>
mod_inverse(std::int64_t a, std::int64_t m) noexcept {
    if (m <= 1) return std::nullopt;
    auto [g, x, y] = ext_gcd(a, m);
    if (g != 1) return std::nullopt;
    std::int64_t inv = x % m;
    return (inv < 0) ? (inv + m) : inv;
}

// Solves x = r_i (mod m_i) for pairwise coprime m_i (product fits in int64_t).
// Input pairs are {remainder, modulus}. Returns {x, N} or std::nullopt.
[[nodiscard]] constexpr std::optional<std::pair<std::int64_t, std::int64_t>>
solve_crt(std::span<const std::pair<std::int64_t, std::int64_t>> congruences) noexcept {
    std::int64_t cur_r = 0;
    std::int64_t cur_m = 1;
    for (auto [rem, mod] : congruences) {
        if (mod <= 0) return std::nullopt;
        std::int64_t target = rem % mod;
        if (target < 0) target += mod;

        // Solve cur_r + cur_m * k = target (mod mod)
        auto inv = mod_inverse(cur_m % mod, mod);
        if (!inv && mod != 1) return std::nullopt; // Moduli are not coprime
        if (mod == 1) continue;

        std::int64_t diff = (target - cur_r % mod + mod) % mod;
        std::int64_t k = (diff * (*inv)) % mod;
        cur_r += cur_m * k;
        cur_m *= mod;
    }
    return std::pair{cur_r, cur_m};
}
\end{lstlisting}

\subsection{Production-Grade \texttt{Modular<Mod>} Value Type}

Wrapping a normalized 64-bit value inside a template class prevents accidental unreduced intermediates. The \texttt{static\_assert} enforces $\text{Mod} \le 2^{32} - 1$ so \texttt{std::uint64\_t} products in \texttt{operator*=} can \emph{never} overflow (for 64-bit moduli up to $10^{18}$, cast through \texttt{\_\_int128} on GCC/Clang).

\begin{lstlisting}[caption={Overflow-Checked C++23 Modular<Mod> Value Class}]
#include <cstdint>
#include <ostream>
#include <stdexcept>
#include <utility>

template <std::int64_t Mod>
class Modular {
    static_assert(Mod > 0, "Modulus must be strictly positive");
    static_assert(Mod <= 4'294'967'295LL,
                  "Mod^2 must fit in uint64_t (use __int128 for 64-bit moduli)");

private:
    std::int64_t value_;

    static constexpr std::int64_t normalize(std::int64_t v) noexcept {
        v %= Mod;
        return (v < 0) ? (v + Mod) : v;
    }

public:
    constexpr Modular() noexcept : value_(0) {}
    constexpr Modular(std::int64_t v) noexcept : value_(normalize(v)) {}

    [[nodiscard]] constexpr std::int64_t get() const noexcept { return value_; }
    [[nodiscard]] static constexpr std::int64_t mod() noexcept { return Mod; }

    // Binary exponentiation: O(log exp) multiplications
    [[nodiscard]] constexpr Modular pow(std::uint64_t exp) const noexcept {
        Modular base = *this;
        Modular result(1);
        while (exp > 0) {
            if (exp & 1ULL) result *= base;
            exp >>= 1;
            if (exp > 0) base *= base;
        }
        return result;
    }

    // Extended Euclidean modular inverse (works for any coprime Mod): O(log Mod)
    [[nodiscard]] constexpr Modular inv() const {
        std::int64_t a = value_, m = Mod;
        std::int64_t u = 0, v = 1;
        while (a != 0) {
            std::int64_t q = m / a;
            m -= q * a; std::swap(a, m);
            u -= q * v; std::swap(u, v);
        }
        if (m != 1) throw std::runtime_error("Modular inverse does not exist");
        return Modular(u);
    }

    constexpr Modular& operator+=(const Modular& rhs) noexcept {
        value_ += rhs.value_;
        if (value_ >= Mod) value_ -= Mod;
        return *this;
    }

    constexpr Modular& operator-=(const Modular& rhs) noexcept {
        value_ -= rhs.value_;
        if (value_ < 0) value_ += Mod;
        return *this;
    }

    constexpr Modular& operator*=(const Modular& rhs) noexcept {
        const auto a = static_cast<std::uint64_t>(value_);
        const auto b = static_cast<std::uint64_t>(rhs.value_);
        const auto m = static_cast<std::uint64_t>(Mod);
        value_ = static_cast<std::int64_t>((a * b) % m);
        return *this;
    }

    constexpr Modular& operator/=(const Modular& rhs) { return *this *= rhs.inv(); }

    friend constexpr Modular operator+(Modular a, const Modular& b) noexcept { return a += b; }
    friend constexpr Modular operator-(Modular a, const Modular& b) noexcept { return a -= b; }
    friend constexpr Modular operator*(Modular a, const Modular& b) noexcept { return a *= b; }
    friend constexpr Modular operator/(Modular a, const Modular& b) { return a /= b; }
    friend constexpr bool operator==(const Modular&, const Modular&) noexcept = default;

    friend std::ostream& operator<<(std::ostream& os, const Modular& m) {
        return os << m.value_;
    }
};

using Mint1000000007 = Modular<1'000'000'007LL>;
using Mint998244353  = Modular<998'244'353LL>;
\end{lstlisting}

\subsection{Case Study: Repunit Divisibility in $O(k)$ Time and $O(1)$ Space}

Using \texttt{std::int64\_t} for \texttt{remainder} prevents 32-bit overflow on \texttt{remainder * 10 + 1} even when $k$ approaches $2 \times 10^9$.

\begin{lstlisting}[caption={Overflow-Safe O(k) Repunit Divisibility via Modular Recurrence}]
#include <cstdint>

// Returns the length of the smallest repunit (1, 11, 111, ...) divisible by k,
// or -1 if none exists.
[[nodiscard]] constexpr int smallestRepunitDivByK(int k) noexcept {
    if (k <= 0 || k % 2 == 0 || k % 5 == 0) {
        return -1; // Repunits end in 1 and cannot be divisible by 2 or 5
    }

    std::int64_t remainder = 0;
    for (int length = 1; length <= k; ++length) {
        remainder = (remainder * 10 + 1) % k;
        if (remainder == 0) return length;
    }
    return -1;
}
\end{lstlisting}

\section{Problem Pattern Recognition}
% ============================================================================

Run through these three checks before picking a modular algorithm.

\begin{calloutbox}{Three-Step Modular Arithmetic Triage}
\begin{enumerate}
    \item \textbf{What is the prompt asking for?}
    \begin{itemize}
        \item ``Return the answer modulo $10^9 + 7$'' $\to$ Reduce every $+, -, \times$ (or use \texttt{Modular<M>}).
        \item ``Position after $k$ steps on a ring/circle'' $\to$ Direct wrap-around \texttt{safe\_mod(idx + k, N)}.
        \item ``Compute $a^b$ for huge $b$'' or ``last $k$ digits of $a^b$'' $\to$ Binary exponentiation modulo $m$ (or $10^k$).
        \item ``Divide inside a modular count'' $\to$ Modular inverse.
        \item ``Simultaneous remainders $x \equiv a_i \pmod{n_i}$'' $\to$ Chinese Remainder Theorem.
    \end{itemize}
    \item \textbf{Check the modulus constraint:}
    \begin{itemize}
        \item Prime $p$ $\to$ Fermat's Little Theorem ($a^{p-2} \bmod p$).
        \item Composite or unknown $m$ $\to$ Extended Euclid (works whenever $\gcd(a, m) = 1$).
        \item Multiple congruences $\to$ Verify pairwise coprimality before applying basic CRT.
    \end{itemize}
    \item \textbf{Check the scale of $b$ and $n$:}
    \begin{itemize}
        \item $n \le 10^6$ (many $\binom{n}{k} \bmod p$) $\to$ $O(n)$ precomputed factorials + inverse factorials (Module 01).
        \item $b \le 10^{18}$ $\to$ Binary exponentiation in $O(\log b)$ ($\le 60$ steps).
        \item $b > 10^{18}$ (e.g., power tower) $\to$ Reduce $b$ modulo $\phi(m)$ via Euler's Theorem (check $\gcd(a, m) = 1$!).
    \end{itemize}
\end{enumerate}
\end{calloutbox}

\section{Glossary}
% ============================================================================

\subsection*{Core Concepts}
\begin{itemize}
    \item \textbf{B\'{e}zout's Identity:} For any integers $a, b$, there exist integers $x, y$ such that $ax + by = \gcd(a, b)$, computed by the Extended Euclidean Algorithm.
    \item \textbf{Binary Exponentiation:} Computing $a^b \bmod m$ in $O(\log b)$ multiplications by decomposing $b$ into binary powers of two and squaring the base iteratively.
    \item \textbf{Chinese Remainder Theorem (CRT):} Guarantees a unique solution modulo $N = \prod n_i$ for a system of congruences with pairwise coprime moduli.
    \item \textbf{Congruence:} The relation $a \equiv b \pmod m$ meaning $a$ and $b$ leave the exact same remainder when divided by $m$.
    \item \textbf{Coprime (Relatively Prime):} Two integers sharing no common positive divisor other than $1$ ($\gcd(a, b) = 1$).
    \item \textbf{Euler's Theorem:} Generalization of Fermat's Little Theorem: $a^{\phi(n)} \equiv 1 \pmod n$ whenever $\gcd(a, n) = 1$.
    \item \textbf{Fermat's Little Theorem:} For prime $p$ and $p \nmid a$, $a^{p-1} \equiv 1 \pmod p$, immediately giving $a^{-1} \equiv a^{p-2} \pmod p$.
    \item \textbf{Modular Multiplicative Inverse:} An integer $x$ with $a \cdot x \equiv 1 \pmod m$; replaces division inside modular arithmetic (exists iff $\gcd(a, m) = 1$).
    \item \textbf{Modulus ($m$ or $n$):} The positive integer at which clock arithmetic wraps back to $0$.
    \item \textbf{Overflow:} Exceeding the maximum value representable by a fixed-width integer type (e.g., $2^{31}-1$ for 32-bit \texttt{int}, $2^{63}-1$ for 64-bit \texttt{int64\_t}).
    \item \textbf{Repunit:} A base-10 integer consisting solely of the digit $1$ ($1, 11, 111, \dots$).
    \item \textbf{Residue Class ($[r]$):} The infinite set of all integers sharing remainder $r$ modulo $m$.
\end{itemize}

\subsection*{Mathematical Symbols}
\begin{itemize}
    \item $a \equiv b \pmod m$: $a$ and $b$ are congruent modulo $m$ ($m \mid (a - b)$).
    \item $m \mid a$ / $m \nmid a$: ``$m$ divides $a$ evenly'' / ``$m$ does not divide $a$''.
    \item $\gcd(a, b)$: Greatest common divisor of $a$ and $b$.
    \item $a^{-1} \bmod m$: Modular multiplicative inverse of $a$ modulo $m$.
    \item $\phi(n)$: Euler's Totient Function---number of integers in $[1, n]$ coprime to $n$.
    \item $\mathbb{Z}/m\mathbb{Z}$: The finite ring of residue classes $\{[0], [1], \dots, [m - 1]\}$ modulo $m$.
    \item $\lfloor x \rfloor$: Floor of $x$ (largest integer $\le x$).
\end{itemize}

\section*{Practice Sheet: Mixed Pen-and-Paper Drills}
\addcontentsline{toc}{section}{Practice Sheet: Mixed Pen-and-Paper Drills}
\markboth{Practice Sheet: Mixed Pen-and-Paper Drills}{}
% ============================================================================

\noindent
Solve these six mixed cold-start drills on scratch paper \emph{without} looking back at the chapter. Each drill tests a different thinking mode with zero hints about which subsection applies.

\begin{tryitbox}{Triage}{Cold-Start Modular Diagnosis}{ex9}
Identify the exact tool and time complexity for each disguised scenario:
\begin{enumerate}
    \item \textbf{Last Three Digits:} Compute the final $3$ decimal digits of $3^{10^{15}}$.
    \item \textbf{Probability Mod Prime:} A DP counts $W$ winning configurations out of $10!$ equally likely permutations and asks for $(W / 10!) \bmod (10^9 + 7)$.
    \item \textbf{Backup Alignment:} Three jobs run every $4$, $9$, and $5$ hours starting at offsets $1$, $2$, and $3$. Find the first hour they all collide.
\end{enumerate}
\end{tryitbox}

\begin{tryitbox}{Trace}{Affine Cipher Key via Extended Euclid}{ex10}
An affine cipher over the 26-letter English alphabet encrypts letter $x$ as $(7x + 3) \bmod 26$. Decrypting requires $7^{-1} \bmod 26$. Compute $7^{-1} \bmod 26$ by hand using the Extended Euclidean Algorithm (forward divisions + back-substitution) and verify your answer.
\end{tryitbox}

\begin{tryitbox}{Debug}{Fermat's Theorem on a Composite Modulus}{ex11}
To compute $3^{-1} \bmod 10$, someone applies Fermat's formula $a^{m - 2} \bmod m$:
\[
    3^{10 - 2} \bmod 10 = 3^8 \bmod 10 = 6{,}561 \bmod 10 = 1.
\]
Show why $1$ is \emph{not} $3^{-1} \bmod 10$, explain which precondition failed, and compute the true inverse $3^{-1} \bmod 10$ using Euler's Theorem ($\phi(10)$) or Extended Euclid.
\end{tryitbox}

\begin{tryitbox}{Twist}{Modular Subtraction and Division Together}{ex12}
Working modulo the prime $M = 11$ with $a = 3$ and $b = 8$, evaluate both:
\[
    (a - b) \bmod 11 \qquad \text{and} \qquad (a / b) \bmod 11,
\]
ensuring both final results lie in $[0, 10]$. Verify your quotient by multiplying back by $b$.
\end{tryitbox}

\begin{tryitbox}{Edge Cases}{Boundary Stress Test}{ex13}
Evaluate or classify each boundary case without guessing:
\begin{enumerate}
    \item $0^0 \bmod 7$ (by the \texttt{mod\_pow} empty-product convention) and $5^{100} \bmod 1$.
    \item Does $6^{-1} \bmod 9$ exist? Why or why not?
    \item What does \texttt{safe\_mod(-14, 7)} return?
\end{enumerate}
\end{tryitbox}

\begin{tryitbox}{Derive}{Discovering Multiplicative Order Modulo $7$}{ex14}
Compute $2^k \bmod 7$ by hand for $k = 1, 2, 3, 4, 5, 6$.
\begin{enumerate}
    \item What is the smallest positive cycle length (order) $d$ where $2^d \equiv 1 \pmod 7$, and how does $d$ relate to $\phi(7) = 6$?
    \item Use that cycle length to compute $2^{1000} \bmod 7$ in two lines.
\end{enumerate}
\end{tryitbox}

\section*{Solutions to Pen-and-Paper Exercises}
\addcontentsline{toc}{section}{Solutions to Pen-and-Paper Exercises}
\markboth{Solutions to Pen-and-Paper Exercises}{}
% ============================================================================

\begin{enumerate}
    \item \label{sol:ex1}\textbf{Exercise 1 (Edge Cases) --- C++ \texttt{\%} vs.\ Math Modulo:}
    $(-1) \bmod 7$: C++ \texttt{-1}, math $\mathbf{6}$;\;
    $(-14) \bmod 7$: C++ \texttt{0}, math $\mathbf{0}$;\;
    $(-17) \bmod 5$: C++ \texttt{-2}, math $\mathbf{3}$;\;
    $(23 - 30) \bmod 11 = (-7) \bmod 11$: C++ \texttt{-7}, math $\mathbf{4}$.
    \trapnote{Writing \texttt{(a \% m) + m} without checking \texttt{r < 0} or taking a final \texttt{\% m}: for \texttt{-14 \% 7 = 0}, adding \texttt{7} returns \texttt{7} instead of \texttt{0}!}

    \item \label{sol:ex2}\textbf{Exercise 2 (Debug) --- Overflow \& Broken Division:}
    \textbf{Bug 1:} \texttt{a * b} $= 10^{18}$ overflows 32-bit \texttt{int} (UB!). Cast to \texttt{std::int64\_t}.
    \textbf{Bug 2:} Raw integer division after \texttt{\% M} is invalid! Multiply by $2^{-1} \bmod M = (M + 1) / 2 = 500{,}000{,}004$.
    True value: $10^9 \equiv -7 \pmod M \implies a \cdot b \equiv (-7)^2 = 49 \pmod M$. Multiply by $2^{-1}$: $(49 \times 500{,}000{,}004) \bmod M = (24 \times M + 500{,}000{,}028) \bmod M = \mathbf{500{,}000{,}028}$ (check: $2 \times 500{,}000{,}028 = 1{,}000{,}000{,}056 = M + 49 \equiv 49$).
    \trapnote{Thinking \texttt{prod / 2} works because the original $ab$ was even---reducing modulo an odd prime $M$ can flip parity ($49$ is odd!), so you still need $2^{-1} \bmod M$.}

    \item \label{sol:ex3}\textbf{Exercise 3 (Trace) --- Binary Exponentiation $3^{13} \bmod 7$:}
    $13 = (1101)_2$. Start $\text{res} = 1, \text{base} = 3$:
    $b_0 = 1 \to \text{res} = 3,\; \text{base} = 9 \equiv 2$;\;
    $b_1 = 0 \to \text{res} = 3,\; \text{base} = 4$;\;
    $b_2 = 1 \to \text{res} = 12 \equiv 5,\; \text{base} = 16 \equiv 2$;\;
    $b_3 = 1 \to \text{res} = 10 \equiv \mathbf{3}$.
    Check: $3^{13} = (3^6)^2 \cdot 3 \equiv 1^2 \cdot 3 = 3 \pmod 7$.
    \trapnote{Reading the bits of $13 = 1101_2$ left-to-right while updating \texttt{base = base * base}---the iterative \texttt{exp >{}>= 1} loop consumes bits right-to-left ($b_0$ first).}

    \item \label{sol:ex4}\textbf{Exercise 4 (Trace) --- $5^{-1} \bmod 11$ Two Ways:}
    \textbf{(1) Extended Euclid:} $11 = 2 \times 5 + 1 \implies 1 = 11 - 2 \times 5$, so $-2 \times 5 \equiv 1 \pmod{11} \implies 5^{-1} \equiv -2 \equiv \mathbf{9} \pmod{11}$.
    \textbf{(2) Fermat:} $5^2 \equiv 3,\; 5^4 \equiv 9,\; 5^8 \equiv 81 \equiv 4 \implies 5^9 = 5^8 \cdot 5 \equiv 20 \equiv \mathbf{9} \pmod{11}$. Check: $5 \times 9 = 45 \equiv 1$.
    \trapnote{Leaving the negative B\'{e}zout coefficient $-2$ unnormalized instead of adding $11$ to get $9 \in [0, 10]$.}

    \item \label{sol:ex5}\textbf{Exercise 5 (Trace) --- Two-Clock CRT:}
    $N = 5 \times 7 = 35$, $M_1 = 7$, $M_2 = 5$.
    Inverses: $7^{-1} \equiv 2^{-1} \equiv 3 \pmod 5$ ($21 \equiv 1$);\; $5^{-1} \equiv 3 \pmod 7$ ($15 \equiv 1$).
    $x \equiv 3(7)(3) + 4(5)(3) = 63 + 60 = 123 \equiv \mathbf{18} \pmod{35}$. Check: $18 \bmod 5 = 3,\; 18 \bmod 7 = 4$.
    \trapnote{Inverting $n_i$ modulo $M_i$ instead of inverting $M_i$ modulo $n_i$---each switch must satisfy $M_i \cdot M_i^{-1} \equiv 1 \pmod{n_i}$.}

    \item \label{sol:ex6}\textbf{Exercise 6 (Debug) --- Broken Euler Reduction on $2^{100} \bmod 12$:}
    \textbf{(1)} $\gcd(2, 12) = 2 \neq 1$, so Euler's shortcut does not apply (also $2^{100}$ is even, so it can never be $\equiv 1 \pmod{12}$!).
    \textbf{(2)} Powers: $2^1 = 2,\; 2^2 = 4,\; 2^3 = 8,\; 2^4 = 16 \equiv 4,\; 2^5 \equiv 8$. Every even exponent $\ge 2$ gives $\mathbf{4}$, so $2^{100} \equiv \mathbf{4} \pmod{12}$.
    \trapnote{Applying $a^B \equiv a^{B \bmod \phi(n)} \pmod n$ without checking $\gcd(a, n) = 1$ first.}

    \item \label{sol:ex7}\textbf{Exercise 7 (Twist) --- Last Two Digits of $7^{2026}$:}
    $\gcd(7, 100) = 1$ and $\phi(100) = 100(1 - \frac{1}{2})(1 - \frac{1}{5}) = 40$, so $7^{2026} \equiv 7^{26} \pmod{100}$. Better yet, $7^2 = 49$ and $7^4 = 49^2 = 2401 \equiv 1 \pmod{100}$, so $7^{26} = (7^4)^6 \cdot 7^2 \equiv 1 \cdot 49 = \mathbf{49} \pmod{100}$.
    \trapnote{Computing $\phi(100) = 99$ by mistaking $100$ for a prime instead of factoring $100 = 2^2 \cdot 5^2 \implies \phi(100) = 40$.}

    \item \label{sol:ex8}\textbf{Exercise 8 (Debug) --- Bugs in \texttt{mod\_pow\_buggy}:}
    \textbf{Bug 1:} Failing to normalize \texttt{a = safe\_mod(a, m)} before the loop leaves negative \texttt{a} returning negative results and large \texttt{a} unreduced.
    \textbf{Bug 2:} For $a = 2 \times 10^9$, \texttt{a * a} $= 4 \times 10^{18}$ is close to signed overflow (and $a = 4 \times 10^9$ overflows signed 64-bit \texttt{int64\_t} immediately on iteration 0!).
    \textbf{Bonus:} When $m = 1, b = 0$, the loop never runs and returns $1$ instead of $1 \bmod 1 = 0$.
    \trapnote{Forgetting to reduce \texttt{base} \emph{before} the first squaring, or assuming \texttt{res = 1} is already reduced when \texttt{m == 1}.}

    \item \label{sol:ex9}\textbf{Exercise 9 (Triage) --- Cold-Start Modular Diagnosis:}
    \textbf{(1)} Binary exponentiation modulo $1{,}000$ in $O(\log 10^{15}) \approx 50$ steps (or reduce exponent mod $\phi(1000) = 400$ first since $\gcd(3, 1000) = 1$).
    \textbf{(2)} Modular inverse of $10!$ modulo prime $10^9 + 7$ via Fermat ($O(\log M)$).
    \textbf{(3)} CRT on pairwise coprime moduli $4, 9, 5$ ($N = 180$) in $O(k \log N)$ time.
    \trapnote{Thinking ``last 3 digits'' needs BigInt string arithmetic instead of simply taking everything modulo $1{,}000$.}

    \item \label{sol:ex10}\textbf{Exercise 10 (Trace) --- Affine Cipher Key $7^{-1} \bmod 26$:}
    Forward: $26 = 3(7) + 5$,\; $7 = 1(5) + 2$,\; $5 = 2(2) + 1$.
    Back-sub: $1 = 5 - 2(2) = 5 - 2(7 - 5) = 3(5) - 2(7) = 3(26 - 3 \cdot 7) - 2(7) = 3(26) - 11(7)$.
    Thus $7^{-1} \equiv -11 \equiv -11 + 26 = \mathbf{15} \pmod{26}$. Check: $7 \times 15 = 105 = 4 \times 26 + 1 \equiv 1$.
    \trapnote{Dropping a minus sign during back-substitution ($-2(-5) = +10$)---always verify $a \cdot x \bmod m = 1$ at the end!}

    \item \label{sol:ex11}\textbf{Exercise 11 (Debug) --- Fermat Fails on Composite $m = 10$:}
    $3 \times 1 = 3 \not\equiv 1 \pmod{10}$, so $1$ is wrong! Fermat's Little Theorem requires a \textbf{prime} modulus, and $10 = 2 \times 5$ is composite.
    Using Euler's Theorem ($\gcd(3, 10) = 1,\; \phi(10) = 4$): $3^{-1} \equiv 3^{\phi(10) - 1} = 3^3 = 27 \equiv \mathbf{7} \pmod{10}$ (check: $3 \times 7 = 21 \equiv 1$).
    \trapnote{Using $a^{m - 2} \bmod m$ blindly for every modulus---for composite $m$ with $\gcd(a, m) = 1$, the true exponent is $\phi(m) - 1$ (or use Extended Euclid).}

    \item \label{sol:ex12}\textbf{Exercise 12 (Twist) --- Subtraction and Division Mod $11$:}
    \textbf{Subtraction:} $(3 - 8) \bmod 11 = -5 \equiv -5 + 11 = \mathbf{6} \pmod{11}$.
    \textbf{Division:} Find $8^{-1} \bmod 11$: $8 \times 7 = 56 = 5 \times 11 + 1 \implies 8^{-1} \equiv 7$. Then $3 \times 7 = 21 \equiv \mathbf{10} \pmod{11}$. Check: $8 \times 10 = 80 = 7 \times 11 + 3 \equiv 3 \pmod{11}$.
    \trapnote{Reporting $-5$ for modular subtraction or dividing $3 / 8 = 0$ in integer arithmetic instead of multiplying by $8^{-1} \bmod 11$.}

    \item \label{sol:ex13}\textbf{Exercise 13 (Edge Cases) --- Boundary Stress Test:}
    \textbf{(1)} $0^0 \bmod 7 = \mathbf{1}$ (empty product, loop runs 0 times); $5^{100} \bmod 1 = \mathbf{0}$ (every integer is divisible by $1$).
    \textbf{(2)} \textbf{Does not exist:} $\gcd(6, 9) = 3 > 1$, so $6x \bmod 9$ only ever hits $\{0, 3, 6\}$ and can never reach $1$.
    \textbf{(3)} \texttt{safe\_mod(-14, 7)} returns \textbf{0} (since \texttt{-14 \% 7 == 0}, which is not $< 0$).
    \trapnote{Returning $1$ for $a^b \bmod 1$ because \texttt{result} starts at $1$, or trying to invert a number sharing a factor with the modulus.}

    \item \label{sol:ex14}\textbf{Exercise 14 (Derive) --- Multiplicative Order of $2 \bmod 7$:}
    Powers: $2^1 \equiv 2,\; 2^2 \equiv 4,\; 2^3 \equiv 1,\; 2^4 \equiv 2,\; 2^5 \equiv 4,\; 2^6 \equiv 1$.
    \textbf{(1)} Order $d = \mathbf{3}$, which evenly divides $\phi(7) = 6$ (a consequence of Lagrange's Theorem).
    \textbf{(2)} $1000 = 3 \times 333 + 1 \implies 2^{1000} = (2^3)^{333} \cdot 2^1 \equiv 1^{333} \cdot 2 = \mathbf{2} \pmod 7$.
    \trapnote{Assuming the shortest cycle always equals $\phi(p) = p - 1$---the true order can be any divisor of $p - 1$, though $a^{p - 1} \equiv 1$ always holds too!}
\end{enumerate}
